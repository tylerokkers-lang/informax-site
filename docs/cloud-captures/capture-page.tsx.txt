const NAV = new Set<NavItem["id"]>(["overview", "spaces", "content", "activity", "team", "settings"]);

/** A fixed, believable moment: Wednesday afternoon, 14:42 in London. */
const NOW = new Date("2026-10-21T13:42:00Z");
const ago = (min: number) => new Date(NOW.getTime() - min * 60_000).toISOString();
const ahead = (min: number) => new Date(NOW.getTime() + min * 60_000).toISOString();
const v = (n: number, name: string, origin: "publish" | "temporary" | "restore", by: string, minAgo: number) =>
  ({ id: `v${n}`, version_no: n, kind: "pdf", source_ref: `private/${n}.pdf`, display_name: name, origin, restored_from_version_id: null, created_by_label: by, created_at: ago(minAgo) });

/** A default that was replaced (or not yet): the earlier versions stay in the history. */
function replaced(names: string[], by: string, updatedName?: string): ContentState {
  const all = names.map((n, i) => v(i + 1, n, "publish", by, (names.length - i) * 60 * 24 * 9));
  if (updatedName) all.push(v(all.length + 1, updatedName, "publish", by, 2));
  const cur = all[all.length - 1];
  const defaults = all.map((x, i) => ({ version_id: x.id, valid_from: x.created_at, valid_to: i < all.length - 1 ? all[i + 1].created_at : null }));
  return {
    versions: [...all].reverse(),
    defaults: defaults.reverse(),
    temporaries: [],
    live: {
      live: { source: "default", kind: "pdf", name: cur.display_name, version: cur.version_no, version_id: cur.id, since: cur.created_at, expires_at: null },
      default: { kind: "pdf", name: cur.display_name, version: cur.version_no, version_id: cur.id, since: cur.created_at, by },
      scheduled: null,
    },
  } as ContentState;
}

/** Temporary content live now, going back to the default by itself. */
function temporary(defName: string, tempName: string, by: string, endsInMin: number): ContentState {
  const d = v(1, defName, "publish", by, 60 * 24 * 20);
  const t = v(2, tempName, "temporary", by, 3);
  return {
    versions: [t, d],
    defaults: [{ version_id: "v1", valid_from: d.created_at, valid_to: null }],
    temporaries: [{ version_id: "v2", starts_at: t.created_at, expires_at: ahead(endsInMin), ended_at: null, end_reason: null }],
    live: {
      live: { source: "temporary", kind: "pdf", name: tempName, version: 2, version_id: "v2", since: t.created_at, expires_at: ahead(endsInMin) },
      default: { kind: "pdf", name: defName, version: 1, version_id: "v1", since: d.created_at, by },
      scheduled: null,
    },
  } as ContentState;
}

const CAN = { manage: true, restore: true, temporary: true, assign: true, end: true };
const tz = "Europe/London";

function ap(i: "hotel" | "restaurant", name: string, area: string, classification: string, state: ContentState, notice?: string) {
  return shell(i, <AccessPointView profile={PROFILES[i]}
    accessPoint={{ id: "ap1", name, status: "active", locationId: i === "hotel" ? "l1" : "r1", classificationId: "c1" }}
    area={{ id: "a1", name: area }} locationName={null}
    code={{ id: "code1", publicCode: "7m4k2x9q0p", url: "https://go.informax.cloud/c/7m4k2x9q0p", status: "active" }}
    classifications={[{ id: "c1", name: classification, status: "active" }]}
    can={{ manage: true, archive: true, bulk: true }}
    notice={notice ? { kind: "ok", text: notice } : undefined}
    content={{ state, timeZone: tz, now: NOW, can: CAN }} />);
}

function overview(i: "hotel" | "restaurant") {
  const now = NOW;
  const sp = (id: string, name: string, parent: string | null) => ({ id, name, parent_id: parent, status: "active" as const });
  const hotel = i === "hotel";
  const spaces = hotel
    // Every area holds at least two Access Points: an area with exactly one triggers the Quick access bug (HANDOVER.md §0c).
    ? [sp("bed", "Bedrooms", null), sp("ct", "Coffee Table", "bed"), sp("bath", "Bathroom", "bed"), sp("bs", "Bedside Table", "bed"), sp("spa", "Spa", null), sp("spa-e", "Spa Entrance", "spa"), sp("spa-r", "Relaxation Room", "spa"), sp("rest", "The Garden Room", null), sp("gr", "Restaurant Menu", "rest"), sp("wl", "Wine List", "rest"), sp("pub", "Public Areas", null), sp("rec", "Reception", "pub"), sp("lob", "Lobby Lounge", "pub")]
    : [sp("main", "Main Restaurant", null), sp("menu", "Restaurant Menu", "main"), sp("drinks", "Drinks Menu", "main"), sp("ter", "Terrace", null), sp("tmenu", "Terrace Menu", "ter"), sp("bar", "Bar", null), sp("wine", "Wine List", "bar")];
  const leaf = spaces.filter((x) => !spaces.some((y) => y.parent_id === x.id));
  const placements = leaf.map((l, n) => ({ space_id: l.id, code_id: `c${n}` }));
  const versions = [
    { id: "v-main", kind: "pdf", display_name: hotel ? "Guest Directory.pdf" : "Dinner menu (hake tonight).pdf" },
    { id: "v-temp", kind: "pdf", display_name: hotel ? "Spa this afternoon.pdf" : "Sunday lunch.pdf" },
    { id: "v-spa", kind: "pdf", display_name: "Spa treatments.pdf" },
  ];
  const tempSpace = hotel ? "bath" : "tmenu";
  const summary = summariseOverview({
    spaces, placements, versions, now, activeCodeIds: new Set(placements.map((p) => p.code_id)),
    defaults: leaf.map((l) => ({ space_id: l.id, version_id: hotel && (l.id === "bath" || l.id === "spa-e" || l.id === "spa-r") ? "v-spa" : "v-main" })),
    temporaries: [{ space_id: tempSpace, version_id: "v-temp", starts_at: ago(20), expires_at: ahead(hotel ? 198 : 100), ended_at: null }],
  });
  const f = hotel ? H : R;
  const by = f.user.name;
  const ev = (id: string, action: string, subject: string, min: number, payload: Record<string, unknown> = {}) =>
    ({ id, organisation_id: "o", location_id: f.loc.id, actor_kind: "user", actor_label: by, action, subject_type: "space", subject_id: "s", subject_label: subject, payload, visibility: "operational", occurred_at: ago(min) });
  const recent = hotel
    ? [ev("e1", "content.temporary_set", "Bathroom", 20, { name: "Spa this afternoon.pdf", expires_at: ahead(198) }), ev("e2", "content.replaced", "Coffee Table", 95, { name: "Guest Directory (new dinner hours).pdf" })]
    : [ev("e1", "content.replaced", "Restaurant Menu", 2, { name: "Dinner menu (hake tonight).pdf" }), ev("e2", "content.temporary_set", "Terrace Menu", 20, { name: "Sunday lunch.pdf", expires_at: ahead(100) })];
  return shell(i, <OverviewView profile={PROFILES[i]} organisationName={f.org} greeting={`Good afternoon, ${by.split(" ")[0]}`} locations={[{ ...f.loc, timezone: tz, status: "active" }]}
    roleLine={PROFILES[i].roleLabels.admin} canInvite summary={{ ...summary, retiredCodes: 0 }} recent={recent as never} timeZone={tz} now={now}
    roleLabel={(k: string) => (PROFILES[i].roleLabels as Record<string, string>)[k] ?? k} />);
}

      return ap("hotel", "Coffee Table", "Bedrooms", "Coffee Table", replaced(["Guest Directory.pdf"], H.user.name, "Guest Directory (new dinner hours).pdf"), "Published. It is live now.");
    case "h-spa-default":

      return ap("hotel", "Bathroom", "Bedrooms", "Bathroom", temporary("Spa treatments.pdf", "Spa this afternoon.pdf", H.user.name, 198), "Temporary content is live.");
